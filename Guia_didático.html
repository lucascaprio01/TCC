projetar3D_sim3 = function(x, y, z, elev_deg, azim_deg) {
  let el = elev_deg * Math.PI / 180;
  let az = azim_deg * Math.PI / 180;
  
  let x_rot = x * Math.cos(az) - y * Math.sin(az);
  let y_rot = x * Math.sin(az) + y * Math.cos(az);
  
  let x_tela = x_rot;
  let y_tela = z * Math.cos(el) + y_rot * Math.sin(el);
  
  return {x_tela, y_tela};
}

dados_orbita_i = angulos_globais.map(function(E) {
  let x_plano = a_fixo_sim3 * Math.cos(E);
  let y_plano = a_fixo_sim3 * Math.sin(E);
  let i_rad = i_graus * Math.PI / 180;
  
  let x3d = x_plano;
  let y3d = y_plano * Math.cos(i_rad);
  let z3d = y_plano * Math.sin(i_rad);
  
  return projetar3D_sim3(x3d, y3d, z3d, elevacao_cam3, azimute_cam3);
})

projetar3D_sim4 = function(x, y, z, elev_deg, azim_deg) {
  let el = elev_deg * Math.PI / 180;
  let az = azim_deg * Math.PI / 180;
  let x_rot = x * Math.cos(az) - y * Math.sin(az);
  let y_rot = x * Math.sin(az) + y * Math.cos(az);
  let x_tela = x_rot;
  let y_tela = z * Math.cos(el) + y_rot * Math.sin(el);

dados_orbita_raan = angulos_globais.map(function(E) {
  let x_plano = a_fixo_sim4 * Math.cos(E);
  let y_plano = a_fixo_sim4 * Math.sin(E);
  
  // 1. Aplica a inclinação fixa (Eixo X)
  let i_rad = i_fixo_sim4 * Math.PI / 180;
  let x_inc = x_plano;
  let y_inc = y_plano * Math.cos(i_rad);
  let z_inc = y_plano * Math.sin(i_rad);
  
  // 2. Aplica o RAAN girando o sistema pelo Eixo Z (Polos)
  let raan_rad = raan_graus * Math.PI / 180;
  let x3d = x_inc * Math.cos(raan_rad) - y_inc * Math.sin(raan_rad);
  let y3d = x_inc * Math.sin(raan_rad) + y_inc * Math.cos(raan_rad);
  let z3d = z_inc;
  
  return projetar3D_sim4(x3d, y3d, z3d, elevacao_cam4, azimute_cam4);
})

  return projetar3D_sim4(x3d, y3d, 0, elevacao_cam4, azimute_cam4);

projetar3D_sim5 = function(x, y, z, elev_deg, azim_deg) {
  let el = elev_deg * Math.PI / 180;
  let az = azim_deg * Math.PI / 180;
  let x_rot = x * Math.cos(az) - y * Math.sin(az);
  let y_rot = x * Math.sin(az) + y * Math.cos(az);
  let x_tela = x_rot;
  let y_tela = z * Math.cos(el) + y_rot * Math.sin(el);
  return {x_tela, y_tela};
}

// Variáveis fixas exclusivas para a simulação 5
a_fixo_sim5 = 5;
e_fixo_sim5 = 0.6;
i_fixo_sim5 = 45;

dados_orbita_arg = angulos_globais.map(function(E) {
  // 1. Órbita no plano 2D padrão
  let x0 = a_fixo_sim5 * Math.cos(E) - (a_fixo_sim5 * e_fixo_sim5);
  let y0 = a_fixo_sim5 * Math.sqrt(1 - Math.pow(e_fixo_sim5, 2)) * Math.sin(E);
  
  // 2. Rotaciona a órbita pelo Argumento do Perigeu (eixo Z do plano orbital)
  let w_rad = arg_perigeu * Math.PI / 180;
  let x1 = x0 * Math.cos(w_rad) - y0 * Math.sin(w_rad);
  let y1 = x0 * Math.sin(w_rad) + y0 * Math.cos(w_rad);
  
  // 3. Aplica a inclinação fixa (gira no eixo X)
  let i_rad = i_fixo_sim5 * Math.PI / 180;
  let x2 = x1;
  let y2 = y1 * Math.cos(i_rad);
  let z2 = y1 * Math.sin(i_rad);
  
  return projetar3D_sim5(x2, y2, z2, elevacao_cam5, azimute_cam5);
})

// Equador para referência
dados_equador_sim5 = angulos_globais.map(function(E) {
  let x3d = (a_fixo_sim5 + 1.5) * Math.cos(E);
  let y3d = (a_fixo_sim5 + 1.5) * Math.sin(E);
  return projetar3D_sim5(x3d, y3d, 0, elevacao_cam5, azimute_cam5);
})

// Linha dos Nodos (Eixo X) - Intersecção entre o equador e a órbita
linha_nodos_sim5 = [
  projetar3D_sim5(-a_fixo_sim5 - 2, 0, 0, elevacao_cam5, azimute_cam5),

]

centro_terra_sim5 = projetar3D_sim5(0, 0, 0, elevacao_cam5, azimute_cam5)

// Linha que aponta da Terra direto para o Perigeu
vetor_perigeu = [
  centro_terra_sim5,
  dados_orbita_arg[0] // O índice 0 (E=0) é sempre o perigeu!
]

dados_simulacao = {
  // Se o botão de reset for clicado, encerra a função e retorna vazio
  if (acao_botoes === "reset") {
    return [];
  }

  // Lemos os valores usando '.value' da raiz do DOM para evitar reatividade automática ao digitar
  let v_tmax = viewof t_max.value;
  let v_inc = viewof inc_graus.value;
  let v_rx = viewof rx.value;
  let v_ry = viewof ry.value;
  let v_rz = viewof rz.value;
  let v_vx = viewof vx.value;
  let v_vy = viewof vy.value;
  let v_vz = viewof vz.value;

  let inc_rad = v_inc * Math.PI / 180;
  
  let r0_x = v_rx;
  let r0_y = (v_ry * Math.cos(inc_rad)) - (v_rz * Math.sin(inc_rad));
  let r0_z = (v_ry * Math.sin(inc_rad)) + (v_rz * Math.cos(inc_rad));

  let v0_x = v_vx;
  let v0_y = (v_vy * Math.cos(inc_rad)) - (v_vz * Math.sin(inc_rad));
  let v0_z = (v_vy * Math.sin(inc_rad)) + (v_vz * Math.cos(inc_rad));

  const mu = 398600.0;

  const f_add = (a, b) => a.map((v, i) => v + b[i]);
  const f_sub = (a, b) => a.map((v, i) => v - b[i]);
  const f_mul = (a, s) => a.map(v => v * s);
  const f_mag = (a) => Math.sqrt(a.reduce((acc, v) => acc + v * v, 0));

  const v_sum = (...vecs) => {
    let res = [...vecs[0]];
    for(let i = 1; i < vecs.length; i++){
      for(let j = 0; j < res.length; j++){
        res[j] += vecs[i][j];
      }
    }
    return res;
  };

  const f_acel = (st) => {
    let r = [st[0], st[1], st[2]];
    let v = [st[3], st[4], st[5]];
    let mag = f_mag(r);
    if (mag === 0) throw new Error("Colisão!");
    
    let fat = -mu / Math.pow(mag, 3);
    let a = f_mul(r, fat);
    return [...v, ...a];
  };

  const step_rk = (y_in, h_in) => {
    let k1 = f_mul(f_acel(y_in), h_in);
    let k2 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 1/4))), h_in);
    let k3 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 3/32), f_mul(k2, 9/32))), h_in);
    let k4 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 1932/2197), f_mul(k2, -7200/2197), f_mul(k3, 7296/2197))), h_in);
    let k5 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 439/216), f_mul(k2, -8), f_mul(k3, 3680/513), f_mul(k4, -845/4104))), h_in);
    let k6 = f_mul(f_acel(v_sum(y_in, f_mul(k1, -8/27), f_mul(k2, 2), f_mul(k3, -3544/2565), f_mul(k4, 1859/4104), f_mul(k5, -11/40))), h_in);

    let w_out = v_sum(y_in, f_mul(k1, 25/216), f_mul(k3, 1408/2565), f_mul(k4, 2197/4104), f_mul(k5, -1/5));
    let q_out = v_sum(y_in, f_mul(k1, 16/135), f_mul(k3, 6656/12825), f_mul(k4, 28561/56430), f_mul(k5, -9/50), f_mul(k6, 2/55));
    
    return { w_out, q_out };
  };

  let tol = 1e-9;
  let state = [r0_x, r0_y, r0_z, v0_x, v0_y, v0_z];
  let t_curr = 0;
  let h_curr = 10.0;
  let max_it = 10000;
  let it = 0;
  
  let res = [];
  
  while (t_curr < v_tmax && it < max_it) {
    let { w_out, q_out } = step_rk(state, h_curr);
    let diff = f_mag(f_sub(w_out, q_out));
    let scale = Math.max(f_mag(state), 1e-8);
    let err = diff / scale;
    
    if (err > tol) {
      h_curr = 0.9 * h_curr * Math.pow(tol / err, 0.2);
      continue;
    }
    
    state = q_out;
    t_curr += h_curr;
    it++;
    
    if (err < 0.1 * tol) h_curr *= 2.0;
    if (t_curr + h_curr > v_tmax) h_curr = v_tmax - t_curr;
    
    res.push({
      t: t_curr,
      x: state[0], y: state[1], z: state[2]
    });
  }
  
  return res;
}

projetar3D_sim5 = function(x, y, z, elev_deg, azim_deg) {
  let el = elev_deg * Math.PI / 180;
  let az = azim_deg * Math.PI / 180;
  let x_rot = x * Math.cos(az) - y * Math.sin(az);
  let y_rot = x * Math.sin(az) + y * Math.cos(az);
  let x_tela = x_rot;
  let y_tela = z * Math.cos(el) + y_rot * Math.sin(el);
  return {x_tela, y_tela};
}

// Variáveis fixas exclusivas para a simulação 5
a_fixo_sim5 = 5;
e_fixo_sim5 = 0.6;
i_fixo_sim5 = 45;

dados_orbita_arg = angulos_globais.map(function(E) {
  // 1. Órbita no plano 2D padrão
  let x0 = a_fixo_sim5 * Math.cos(E) - (a_fixo_sim5 * e_fixo_sim5);
  let y0 = a_fixo_sim5 * Math.sqrt(1 - Math.pow(e_fixo_sim5, 2)) * Math.sin(E);
  
  // 2. Rotaciona a órbita pelo Argumento do Perigeu (eixo Z do plano orbital)
  let w_rad = arg_perigeu * Math.PI / 180;
  let x1 = x0 * Math.cos(w_rad) - y0 * Math.sin(w_rad);
  let y1 = x0 * Math.sin(w_rad) + y0 * Math.cos(w_rad);
  
  // 3. Aplica a inclinação fixa (gira no eixo X)
  let i_rad = i_fixo_sim5 * Math.PI / 180;
  let x2 = x1;
  let y2 = y1 * Math.cos(i_rad);
  let z2 = y1 * Math.sin(i_rad);
  
  return projetar3D_sim5(x2, y2, z2, elevacao_cam5, azimute_cam5);
})

  let r0_y = (v_ry * Math.cos(inc_rad)) - (v_rz * Math.sin(inc_rad));
  let r0_z = (v_ry * Math.sin(inc_rad)) + (v_rz * Math.cos(inc_rad));

  let v0_x = v_vx;
  let v0_y = (v_vy * Math.cos(inc_rad)) - (v_vz * Math.sin(inc_rad));
  let v0_z = (v_vy * Math.sin(inc_rad)) + (v_vz * Math.cos(inc_rad));

  const mu = 398600.0;

  const f_add = (a, b) => a.map((v, i) => v + b[i]);
  const f_sub = (a, b) => a.map((v, i) => v - b[i]);
  const f_mul = (a, s) => a.map(v => v * s);
  const f_mag = (a) => Math.sqrt(a.reduce((acc, v) => acc + v * v, 0));

  const v_sum = (...vecs) => {
    let res = [...vecs[0]];
    for(let i = 1; i < vecs.length; i++){
      for(let j = 0; j < res.length; j++){
        res[j] += vecs[i][j];
      }
    }
    return res;
  };

  const f_acel = (st) => {
    let r = [st[0], st[1], st[2]];
    let v = [st[3], st[4], st[5]];
    let mag = f_mag(r);
    if (mag === 0) throw new Error("Colisão!");
    
    let fat = -mu / Math.pow(mag, 3);
    let a = f_mul(r, fat);
    return [...v, ...a];
  };

  const step_rk = (y_in, h_in) => {
    let k1 = f_mul(f_acel(y_in), h_in);
    let k2 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 1/4))), h_in);
    let k3 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 3/32), f_mul(k2, 9/32))), h_in);
    let k4 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 1932/2197), f_mul(k2, -7200/2197), f_mul(k3, 7296/2197))), h_in);
    let k5 = f_mul(f_acel(v_sum(y_in, f_mul(k1, 439/216), f_mul(k2, -8), f_mul(k3, 3680/513), f_mul(k4, -845/4104))), h_in);
    let k6 = f_mul(f_acel(v_sum(y_in, f_mul(k1, -8/27), f_mul(k2, 2), f_mul(k3, -3544/2565), f_mul(k4, 1859/4104), f_mul(k5, -11/40))), h_in);

    let w_out = v_sum(y_in, f_mul(k1, 25/216), f_mul(k3, 1408/2565), f_mul(k4, 2197/4104), f_mul(k5, -1/5));
    let q_out = v_sum(y_in, f_mul(k1, 16/135), f_mul(k3, 6656/12825), f_mul(k4, 28561/56430), f_mul(k5, -9/50), f_mul(k6, 2/55));
    
    return { w_out, q_out };
  };

  let tol = 1e-9;
  let state = [r0_x, r0_y, r0_z, v0_x, v0_y, v0_z];
  let t_curr = 0;
  let h_curr = 10.0;
  let max_it = 10000;
  let it = 0;
  
  let res = [];
  
  while (t_curr < v_tmax && it < max_it) {
    let { w_out, q_out } = step_rk(state, h_curr);
    let diff = f_mag(f_sub(w_out, q_out));
    let scale = Math.max(f_mag(state), 1e-8);
    let err = diff / scale;
    
    if (err > tol) {
      h_curr = 0.9 * h_curr * Math.pow(tol / err, 0.2);
      continue;
    }
    
    state = q_out;
    t_curr += h_curr;
    it++;
    
    if (err < 0.1 * tol) h_curr *= 2.0;
    if (t_curr + h_curr > v_tmax) h_curr = v_tmax - t_curr;
    
    res.push({
      t: t_curr,
      x: state[0], y: state[1], z: state[2]
    });
  }
  
  return res;
}
